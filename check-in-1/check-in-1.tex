\section{Language grammar (EBNF)}
We use EBNF\cite{ISO14977} for the following grammar definition:
  \begin{lstlisting}
    (* Program *)
    program = { function_definition }, { expression } ;

    (* Functions *)
    function_definition = "fn", identifier, "(", [ identifier, { ",", identifier } ], ")",
    "{", { expression }, "}" ;

    function_call = identifier, "(", [ expression, { ",", expression } ], ")" ;

    (* Expressions *)
    expression = variable | integer | string | assignment
    | loop | function_call | component ;

    assignment = variable, ":=", expression ;

    loop = "repeat", ( variable | integer ),
    "{", { expression }, "}" ;

    (* UI Components *)
    component = component_type, "{", { ( component | property ) }, "}"  ;

    component_type = "card" | "button" | "text" | "column" | "row"  ;

    property = identifier, ":", expression ;

    (* Identifiers and Variables *)
    variable = identifier ;

    identifier = ? [a-zA-Z0-9]+ ? ;

    (* Primitive Types *)
    integer = ? [0-9]+ ? ;

    string = '"', { string_char | escape }, '"'  ;

    string_char = ? any Unicode character except " and \ ?  ;

    escape = "\", ( '"' | "\" | "n" | "t" )  ;
  \end{lstlisting}

\section{Operational semantics}
We define the semantic relations as

\[
  Functions \times Store \times Expr \rightharpoonup Functions \times
  Store \times UI
\]

\noindent with

\[
  \begin{gathered}
    \phi \in Functions\\
    \sigma \in Store\\
    e \in Expr
  \end{gathered}
\]

\subsection{Primitives}

\[
  \frac{}
  {\langle \sigma, n\rangle \Downarrow \langle \sigma, n\rangle}
  \quad\textsc{Int}
\]

\[
  \frac{v = \sigma(x)}
  {\langle \sigma, x\rangle \Downarrow \langle \sigma, v\rangle}
  \quad\textsc{Var}
\]

\[
  \frac{
    \langle \sigma,e\rangle \Downarrow \langle \sigma',v\rangle
  }{
    \langle \sigma, x := e\rangle
    \Downarrow
    \langle \sigma'[x\mapsto v]\rangle
  }
  \quad\textsc{Assign}
\]
\subsection{Sequence}
\[
  \frac{
    \langle \sigma,e_1\rangle\Downarrow\langle \sigma',v_1\rangle
    \qquad
    \langle \sigma',e_2\rangle\Downarrow\langle \sigma'',v_2\rangle
  }{
    \langle \sigma,e_1;e_2\rangle
    \Downarrow\langle \sigma'',v_2\rangle
  }
  \quad\textsc{Seq}
\]
\subsection{Functions \cite[Section~8.1]{hong2023programming}}
Here, $f$ defines \textit{function name}, $x_1, ..., x_n$ are
function arguments and $e_f$ denotes function body.
\subsubsection{Function definition}
\[
  \frac{}{
    \langle
    \phi,\operatorname{fn}\ f(x_1,\ldots,x_n)\{e_f\}
    \rangle
    \Downarrow
    \phi[f\mapsto((x_1,\ldots,x_n),e_f)]
  }
  \quad\textsc{Def}
\]
\subsubsection{Function call}
\[
  \frac{
    \begin{gathered}
      \phi(f)=(x_1,\ldots,x_n,e_f)
      \quad
      \langle\sigma,e_i\rangle\Downarrow\langle\sigma',v_i\rangle
      \quad
      \langle\phi, \sigma,e_f[v_1/x_1,\ldots,v_n/x_n]\rangle
      \Downarrow\langle\sigma',v\rangle
    \end{gathered}
  }{
    \langle\phi,\sigma,f(e_1,\ldots,e_n)\rangle
    \Downarrow\langle\phi,\sigma',v\rangle
  }
  \quad\textsc{Call}
\]
